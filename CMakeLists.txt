cmake_minimum_required(VERSION 3.25)
project(skate_recomp LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
    message(FATAL_ERROR "Build the Port with LLVM Clang on Windows.")
endif()

find_package(rexglue 0.10.1 EXACT CONFIG REQUIRED)
include(CTest)
add_subdirectory(tools)

add_library(skate_host OBJECT src/skate_app.cpp src/kernel_audit.cpp
    src/vector_bounds.cpp src/vector_minmax.cpp src/input_timing.cpp src/scalar_guards.cpp)
target_compile_features(skate_host PRIVATE cxx_std_23)
target_link_libraries(skate_host PRIVATE rex::runtime)
rexglue_apply_target_settings(skate_host)

if(BUILD_TESTING)
    foreach(vector_test IN ITEMS bounds minmax)
        set(test_target "test_vector_${vector_test}")
        add_executable(${test_target} "tools/${test_target}.cpp" "src/vector_${vector_test}.cpp")
        target_compile_features(${test_target} PRIVATE cxx_std_23)
        target_include_directories(${test_target} PRIVATE src)
        target_link_libraries(${test_target} PRIVATE rex::runtime)
        rexglue_apply_target_settings(${test_target})
        add_test(NAME "vector_${vector_test}" COMMAND ${test_target})
        set_tests_properties("vector_${vector_test}" PROPERTIES TIMEOUT 30)
    endforeach()
    add_executable(test_input_timing tools/test_input_timing.cpp src/input_timing.cpp)
    target_compile_features(test_input_timing PRIVATE cxx_std_23)
    target_include_directories(test_input_timing PRIVATE src)
    target_link_libraries(test_input_timing PRIVATE rex::runtime)
    rexglue_apply_target_settings(test_input_timing)
    add_test(NAME input_timing COMMAND test_input_timing)
    set_tests_properties(input_timing PROPERTIES TIMEOUT 30)
    add_executable(test_scalar_guards tools/test_scalar_guards.cpp src/scalar_guards.cpp)
    target_compile_features(test_scalar_guards PRIVATE cxx_std_23)
    target_include_directories(test_scalar_guards PRIVATE src)
    target_link_libraries(test_scalar_guards PRIVATE rex::runtime)
    rexglue_apply_target_settings(test_scalar_guards)
    add_test(NAME scalar_guards COMMAND test_scalar_guards)
    set_tests_properties(scalar_guards PROPERTIES TIMEOUT 30)
endif()

set(SKATE_GENERATED_DIR "" CACHE PATH "Directory containing locally recompiled skate. code")
if(SKATE_GENERATED_DIR)
    include("${SKATE_GENERATED_DIR}/sources.cmake")
    add_library(skate_guest OBJECT ${GENERATED_SOURCES})
    target_compile_features(skate_guest PRIVATE cxx_std_23)
    target_include_directories(skate_guest PRIVATE "${SKATE_GENERATED_DIR}")
    target_link_libraries(skate_guest PRIVATE rex::runtime)
    target_compile_options(skate_guest PRIVATE -fasync-exceptions -O1)
    target_precompile_headers(skate_guest PRIVATE "${SKATE_GENERATED_DIR}/skateport_pch.h")
    rexglue_apply_target_settings(skate_guest)

    add_executable(skateport WIN32 src/recompiled_guest.cpp)
    target_compile_features(skateport PRIVATE cxx_std_23)
    target_include_directories(skateport PRIVATE "${SKATE_GENERATED_DIR}")
    target_link_libraries(skateport PRIVATE skate_host skate_guest rex::runtime)
    rexglue_configure_target(skateport GPU_PLUGINS xenos)
endif()
